```javascript Node.js com Axios
// flowpay.js
const axios = require('axios');

const BASE_URL = 'https://flowapplications.com.br';
const API_TOKEN = process.env.FLOW_API_KEY || 'seu_token_aqui';

// Criar instância do axios com configurações padrão
const flowpay = axios.create({
  baseURL: BASE_URL,
  timeout: 30000,
  headers: {
    'Authorization': `Bearer ${API_TOKEN}`,
    'Content-Type': 'application/json'
  }
});

// Interceptor para tratamento de erros
flowpay.interceptors.response.use(
  response => response,
  error => {
    if (error.response) {
      console.error('Erro da API:', error.response.data);
    } else if (error.request) {
      console.error('Erro de conexão:', error.message);
    }
    return Promise.reject(error);
  }
);

module.exports = flowpay;
```

```javascript TypeScript
// flowpay.ts
import axios, { AxiosInstance, AxiosError } from 'axios';

const BASE_URL = 'https://flowapplications.com.br';
const API_TOKEN = process.env.FLOW_API_KEY || 'sua_api_key_aqui';

interface CreatePaymentRequest {
  value: number;
  description?: string;
  expiresIn?: number;
  customer?: {
    name?: string;
    email?: string;
    phone?: string;
  };
}

interface PaymentCharge {
  id: string;
  value: number;
  status: 'ACTIVE' | 'PAID' | 'CONFIRMED' | 'COMPLETED' | 'EXPIRED' | 'CANCELLED';
  pixCode: string;
  pixCodeBase64: string;
  qrCode: {
    emv: string;
    image: string;
  };
  publicPaymentUrl: string;
  expiresAt: string;
  createdAt: string;
}

interface PaymentResponse {
  success: boolean;
  charge?: PaymentCharge;
  error?: string;
}

class FlowPayAPI {
  private client: AxiosInstance;

  constructor(token: string = API_TOKEN) {
    this.client = axios.create({
      baseURL: BASE_URL,
      timeout: 30000,
      headers: {
        'Authorization': `Bearer ${token}`,
        'Content-Type': 'application/json'
      }
    });

    this.client.interceptors.response.use(
      response => response,
      (error: AxiosError) => {
        if (error.response) {
          console.error('Erro da API:', error.response.data);
        }
        return Promise.reject(error);
      }
    );
  }

  async createPayment(data: CreatePaymentRequest): Promise<PaymentResponse> {
    try {
      const response = await this.client.post<PaymentResponse>('/api/flow-api/charge', data);
      return response.data;
    } catch (error: any) {
      throw new Error(error.response?.data?.error || 'Erro ao criar pagamento');
    }
  }

  async verifyPayment(chargeId: string): Promise<PaymentResponse> {
    try {
      const response = await this.client.get<PaymentResponse>(`/api/flow-api/charge/${chargeId}`);
      return response.data;
    } catch (error: any) {
      throw new Error(error.response?.data?.error || 'Erro ao verificar pagamento');
    }
  }
}

export default FlowPayAPI;
```

```javascript JavaScript Puro
// criarPagamento.js
const flowpay = require('./flowpay');

async function criarPagamento(valor, descricao = null) {
  const data = {
    value: valor
  };

  if (descricao) {
    data.description = descricao;
  }

  try {
    const response = await flowpay.post('/api/flow-api/charge', data);
    const { success, charge, error } = response.data;

    if (success && charge) {
      console.log('✅ Pagamento criado com sucesso!');
      console.log('🆔 Charge ID:', charge.id);
      console.log('🔲 QR Code:', charge.qrCode.image);
      console.log('📋 PIX Code:', charge.pixCode);
      console.log('🔗 Link Público:', charge.publicPaymentUrl);
      return charge;
    } else {
      throw new Error(error || 'Erro desconhecido');
    }
  } catch (error) {
    console.error('❌ Erro ao criar pagamento:', error.message);
    throw error;
  }
}

// Exemplo de uso
(async () => {
  try {
    const pagamento = await criarPagamento(
      10.50,
      'Pagamento do pedido #123'
    );
    console.log('Pagamento:', pagamento);
  } catch (error) {
    console.error('Erro:', error);
```

```javascript Express.js
// routes/payment.js
const express = require('express');
const router = express.Router();
const flowpay = require('../flowpay');

// Criar pagamento
router.post('/create', async (req, res) => {
  try {
    const { value, description, expiresIn, customer } = req.body;

    // Validação
    if (!value || value <= 0) {
      return res.status(400).json({
        success: false,
        error: 'value é obrigatório e deve ser maior que zero'
      });
    }

    const data = {
      value,
      ...(description && { description }),
      ...(expiresIn && { expiresIn }),
      ...(customer && { customer })
    };

    const response = await flowpay.post('/api/flow-api/charge', data);
    const { success, charge, error } = response.data;

    if (success && charge) {
      res.status(200).json({
        success: true,
        charge: {
          id: charge.id,
          value: charge.value,
          status: charge.status,
          pixCode: charge.pixCode,
          qrCode: charge.qrCode,
          publicPaymentUrl: charge.publicPaymentUrl
        }
      });
    } else {
      res.status(400).json({ success: false, error });
    }
  } catch (error) {
    console.error('Erro ao criar pagamento:', error);
    res.status(500).json({
      success: false,
      error: 'Erro interno do servidor'
    });
  }
});

module.exports = router;
```

```javascript NestJS
// payment.service.ts
import { Injectable, HttpException, HttpStatus } from '@nestjs/common';
import { HttpService } from '@nestjs/axios';
import { firstValueFrom } from 'rxjs';

@Injectable()
export class PaymentService {
  private readonly baseURL = 'https://flowapplications.com.br';
  private readonly apiToken = process.env.FLOW_API_KEY;

  constructor(private httpService: HttpService) {}

  async createPayment(data: {
    value: number;
    description?: string;
    expiresIn?: number;
    customer?: any;
  }) {
    try {
      const response = await firstValueFrom(
        this.httpService.post(
          `${this.baseURL}/api/flow-api/charge`,
          data,
          {
            headers: {
              Authorization: `Bearer ${this.apiToken}`,
              'Content-Type': 'application/json'
            }
          }
        )
      );

      return response.data;
    } catch (error) {
      throw new HttpException(
        error.response?.data?.error || 'Erro ao criar pagamento',
        error.response?.status || HttpStatus.INTERNAL_SERVER_ERROR
      );
    }
  }

  async verifyPayment(chargeId: string) {
    try {
      const response = await firstValueFrom(
        this.httpService.get(
          `${this.baseURL}/api/flow-api/charge/${chargeId}`,
          {
            headers: {
              Authorization: `Bearer ${this.apiToken}`
            }
          }
        )
      );

      return response.data;
    } catch (error) {
      throw new HttpException(
        error.response?.data?.error || 'Erro ao verificar pagamento',
        error.response?.status || HttpStatus.NOT_FOUND
      );
    }
  }
}
```

## Verificar Pagamento

<CodeGroup>

```javascript Função Simples
// verificarPagamento.js
const flowpay = require('./flowpay');

async function verificarPagamento(chargeId) {
  try {
    const response = await flowpay.get(`/api/flow-api/charge/${chargeId}`);
    const { success, charge, error } = response.data;

    if (success && charge) {
      const status = charge.status || charge.paymentMethods?.pix?.status;
      console.log(`📊 Status do pagamento: ${status}`);
      console.log(`💰 Valor: R$ ${charge.value.toFixed(2)}`);
      
      const isPaid = status === 'PAID' || status === 'CONFIRMED' || status === 'COMPLETED';
      
      if (isPaid) {
        console.log('✅ Pagamento confirmado!');
      } else if (status === 'EXPIRED') {
        console.log('⏰ Pagamento expirado');
      } else {
        console.log('⏳ Aguardando pagamento...');
      }
      
      return charge;
    } else {
      throw new Error(error || 'Charge não encontrada');
    }
  } catch (error) {
    console.error('❌ Erro ao verificar pagamento:', error.message);
    throw error;
  }
}

// Exemplo de uso
(async () => {
  try {
    const status = await verificarPagamento('flow_123_1704067200000_abc123');
    console.log('Status:', status);
  } catch (error) {
    console.error('Erro:', error);
  }
})();
```

```javascript Polling (Verificação Periódica)
// verificarPeriodicamente.js
const flowpay = require('./flowpay');

async function verificarPagamento(chargeId) {
  try {
    const response = await flowpay.get(`/api/flow-api/charge/${chargeId}`);
    return response.data;
  } catch (error) {
    throw error;
  }
}

async function aguardarPagamento(chargeId, intervalo = 5000, timeout = 300000) {
  const inicio = Date.now();

  return new Promise((resolve, reject) => {
    const interval = setInterval(async () => {
      try {
        const resultado = await verificarPagamento(chargeId);
        
        if (resultado.success && resultado.charge) {
          const { status } = resultado.charge;

          if (status === 'PAID' || status === 'CONFIRMED' || status === 'COMPLETED') {
            clearInterval(interval);
            resolve(resultado.charge);
          } else if (status === 'EXPIRED') {
            clearInterval(interval);
            reject(new Error('Pagamento expirado'));
          }
        }

        // Verifica timeout
        if (Date.now() - inicio > timeout) {
          clearInterval(interval);
          reject(new Error('Timeout ao aguardar pagamento'));
        }
      } catch (error) {
        clearInterval(interval);
        reject(error);
      }
    }, intervalo);
  });
}

// Exemplo de uso
(async () => {
  try {
    console.log('⏳ Aguardando pagamento...');
    const pagamento = await aguardarPagamento('pedido-123');
    console.log('✅ Pagamento confirmado!', pagamento);
  } catch (error) {
    console.error('❌ Erro:', error.message);
  }
})();
```

## Exemplo Completo - E-commerce

```javascript
// ecommerce-example.js
const express = require('express');
const app = express();
const flowpay = require('./flowpay');

app.use(express.json());

// Criar pagamento para um pedido
app.post('/pedidos/:id/pagamento', async (req, res) => {
  try {
    const { id } = req.params;
    const { valor, cliente } = req.body;

    // Criar pagamento PIX
    const response = await flowpay.post('/payment/create', {
      value: valor,
      comment: `Pagamento do pedido #${id}`,
      expiresIn: 3600, // 1 hora
      customer: cliente ? {
        name: cliente.nome,
        email: cliente.email,
        phone: cliente.telefone,
        taxID: cliente.cpf ? {
          taxID: cliente.cpf,
          type: 'BR:CPF'
        } : undefined
      } : undefined
    });

    if (response.data.success) {
      res.json({
        success: true,
        pagamento: {
          id: response.data.charge.id,
          qrCode: response.data.charge.qrCode.image,
          pixCode: response.data.charge.pixCode,
          link: response.data.charge.publicPaymentUrl,
          expiraEm: response.data.charge.expiresDate
        }
      });
    }
  } catch (error) {
    res.status(500).json({
      success: false,
      error: error.message
    });
  }
});

// Webhook para verificar status
app.post('/webhook/pagamento', async (req, res) => {
  try {
    const { chargeId } = req.body;
    
    const response = await flowpay.get(`/api/flow-api/charge/${chargeId}`);
    
    const status = response.data.charge?.status || response.data.charge?.paymentMethods?.pix?.status;
    const isPaid = status === 'PAID' || status === 'CONFIRMED' || status === 'COMPLETED';
    
    if (response.data.success && isPaid) {
      // Atualizar pedido como pago no banco de dados
      console.log(`✅ Charge ${chargeId} pago!`);
      // await atualizarPedido(chargeId, 'pago');
    }

    res.json({ success: true });
  } catch (error) {
    console.error('Erro no webhook:', error);
    res.status(500).json({ success: false });
  }
});

app.listen(3000, () => {
  console.log('🚀 Servidor rodando na porta 3000');
});
```

## Tratamento de Erros

```javascript
// error-handler.js
class FlowPayError extends Error {
  constructor(message, statusCode, response) {
    super(message);
    this.name = 'FlowPayError';
    this.statusCode = statusCode;
    this.response = response;
  }
}

async function criarPagamentoSeguro(data) {
  try {
    const response = await flowpay.post('/api/flow-api/charge', data);
    return response.data;
  } catch (error) {
    if (error.response) {
      // Erro da API
      const { status, data } = error.response;
      throw new FlowPayError(
        data.error || 'Erro ao criar pagamento',
        status,
        data
      );
    } else if (error.request) {
      // Erro de rede
      throw new FlowPayError(
        'Erro de conexão com a API',
        0,
        null
      );
    } else {
      // Outro erro
      throw new FlowPayError(
        error.message || 'Erro desconhecido',
        0,
        null
      );
    }
  }
}
```

## Variáveis de Ambiente

Crie um arquivo `.env`:

```bash
FLOW_API_KEY=sua_api_key_aqui
FLOW_API_BASE_URL=https://flowapplications.com.br
```

E use com `dotenv`:

```javascript
require('dotenv').config();

const API_KEY = process.env.FLOW_API_KEY;
const BASE_URL = process.env.FLOW_API_BASE_URL;
```

<Note>
Para produção, sempre use variáveis de ambiente para armazenar o token da API. Nunca commite o token no código fonte.
</Note>
